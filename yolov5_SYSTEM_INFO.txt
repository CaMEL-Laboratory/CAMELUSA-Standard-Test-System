#显卡驱动版本
TX2 的 GPU 是直接集成在 SoC 芯片上的，它的驱动与 JetPack 系统的 L4T 内核直接绑定。所以，你无法通过 nvidia-smi 查看版本。
查看系统和 L4T 内核版本（即相当于驱动版本） 的命令：
# 查看 JetPack L4T 版本
head -n 1 /etc/nv_tegra_release
echo# R32 (release), REVISION: 7.6, GCID: 38171779, BOARD: t186ref, EABI: aarch64, DATE: Tue Nov  5 07:49:02 UTC 2024


#CUDA 版本
/usr/local/cuda/bin/nvcc --version
echo#
nvcc: NVIDIA (R) Cuda compiler driver
Copyright (c) 2005-2021 NVIDIA Corporation
Built on Sun_Feb_28_22:34:44_PST_2021
Cuda compilation tools, release 10.2, V10.2.300
Build cuda_10.2_r440.TC440_70.29663091_0

#cuDNN 版本
python -c "import torch; print(torch.backends.cudnn.version())"或者
cat /usr/include/cudnn_version.h | grep CUDNN_MAJOR -A 2
echo#
8201

#PyTorch 版本
# 先确保你在 yolo5_env 环境下
python -c "import torch; print(torch.__version__)"
echo#
1.10.0

sudo jetson_clocks --show   # 或者直接运行下面这行
apt-cache show nvidia-jetpack
